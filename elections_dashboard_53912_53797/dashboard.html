naoPartidos = ["Inscritos", "Votantes", "Brancos", "Nulos", "Votos Val. Exp. (VVE)"]

// Converte o nome do GeoJSON (AVEIRO) para o nome do JSON (Aveiro)
function nomeDistrito(nome) {
  return Object.keys(results.Inscritos).find(d => d.toUpperCase() === nome.toUpperCase());
}

// Votos de cada partido num distrito, do maior para o menor
function votosPorDistrito(distritoNome) {
  const nome = nomeDistrito(distritoNome);
  const result = [];

  for (const [partido, distritos] of Object.entries(results)) {
    if (!naoPartidos.includes(partido)) {
      const votos = distritos[nome];
      if (votos !== undefined) {
        result.push({ partido, votos: +votos || 0 });
      }
    }
  }

  return result.sort((a, b) => b.votos - a.votos);
}

// Mapa principal
mainMap = {
  const mapDiv = html`<div style="height: 100%; min-height: 400px;"></div>`;

  const map = L.map(mapDiv, {
    zoomControl: false,
    dragging: false,
    scrollWheelZoom: false
  });

  map.setView([39.5, -8], 6);

  L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png").addTo(map);

  L.geoJSON(geo, {
    style: (feat) => {
      const distrito = feat.properties.Distrito;
      const vencedor = votosPorDistrito(distrito)[0]?.partido;
      return {
        color: "#4b4b4b",
        weight: 1,
        fillColor: getPartyColors()[vencedor] || "#b3afafab",
        fillOpacity: 0.7
      };
    },
    onEachFeature: (feat, layer) => {
      const nome = nomeDistrito(feat.properties.Distrito);
      const vencedor = votosPorDistrito(nome)[0];
      layer.bindTooltip(`${nome}<br>Partido mais votado: ${vencedor.partido}<br>Votos: ${vencedor.votos.toLocaleString("pt-PT")}`, { sticky: true });
    }
  }).addTo(map);

  setTimeout(() => map.invalidateSize(), 0);

  return mapDiv;
}
